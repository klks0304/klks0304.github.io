// ========== Start mission ==========
function startMission() {
  document.getElementById('birth-result').classList.add('hidden');
  document.getElementById('game-screen').classList.remove('hidden');

  document.getElementById('role-display').innerHTML = state.isCommander
    ? `<span class="role-badge commander">Commander</span>`
    : `<span class="role-badge member">Crew Member</span>`;

  document.getElementById('player-info').textContent =
    `${state.profession.name} · ${state.location}`;

  state.running = true;
  renderBars();
  renderFocusButtons();
  addLog(`Mission started! You are the ${state.isCommander ? "Commander" : "Crew Member"} (${state.profession.name})`);
  state.interval = setInterval(tick, 1000);
}

// ========== Core logic ==========
function tick() {
  if (state.paused || !state.running) return;
  for (let i = 0; i < state.speed; i++) {
    advanceOneSol();
    if (!state.running) break;
  }
  document.getElementById('sol').textContent = state.sol;
  document.getElementById('speed-label').textContent = ` · ${state.speed}×`;
  renderBars();
}

function advanceOneSol() {
  state.sol++;

  // Base consumption
  state.resources.oxygen -= 1.6;
  state.resources.power  -= 1.1;
  state.resources.food   -= 1.4;
  state.resources.water  -= 1.2;
  state.resources.shield -= 0.35;
  state.resources.morale -= 0.25;

  // Focus effect
  if (state.focus) {
    const f = focuses.find(x => x.id === state.focus);
    if (f) {
      for (const [k, v] of Object.entries(f.effect)) {
        state.resources[k] = Math.min(100, Math.max(0, state.resources[k] + v));
      }
    }
  }

  // Random event (~15%)
  if (Math.random() < 0.15) triggerRandomEvent();

  // Failure checks
  for (const [k, v] of Object.entries(state.resources)) {
    if (v <= 0) {
      endMission(false, `Critical system “${k}” has collapsed!`);
      return;
    }
  }
  if (state.commanderMistakes >= 3) {
    endMission(false, "The Commander has made too many poor decisions. Mission failed.");
    return;
  }
  if (state.sol >= state.target) {
    endMission(true, "Successfully maintained the outpost for the target number of Sols!");
  }
}

function chooseFocus(id) {
  state.focus = id;
  const f = focuses.find(x => x.id === id);
  addLog(`Focus changed to: ${f.name}`);
  renderFocusButtons();
}

function showEvent(ev) {
  state.paused = true;
  const area = document.getElementById('event-area');
  area.classList.remove('hidden');

  // Simulated voting (good options tend to get more votes)
  let voteA = Math.floor(Math.random() * 3) + (ev.options[0].good ? 3 : 1);
  let voteB = 7 - voteA;
  if (voteB < 1) { voteA = 5; voteB = 2; }

  let html = `
    <div class="event">
      <strong>${ev.title}</strong>
      <p>${ev.desc}</p>
  `;

  if (state.isCommander) {
    html += `
      <div class="vote-result">
        Crew vote results:<br>
        “${ev.options[0].text}” → ${voteA} votes<br>
        “${ev.options[1].text}” → ${voteB} votes
      </div>
      <p style="font-size:0.9rem;margin-bottom:8px">You are the Commander. Make the final decision:</p>
    `;
  } else {
    html += `<p style="font-size:0.9rem;margin-bottom:8px">You are a crew member. Please vote:</p>`;
  }

  html += `<div style="display:flex;flex-direction:column;gap:8px">`;
  ev.options.forEach((opt, i) => {
    html += `<button onclick="resolveEvent(${i}, ${ev.lifeCritical})">${opt.text}</button>`;
  });
  html += `</div></div>`;

  area.innerHTML = html;
  window._currentEvent = ev;
  window._votes = [voteA, voteB];
}

function resolveEvent(idx, lifeCritical) {
  const ev = window._currentEvent;
  const choice = ev.options[idx];
  const majorityIdx = window._votes[0] >= window._votes[1] ? 0 : 1;

  for (const [k, v] of Object.entries(choice.effect)) {
    state.resources[k] = Math.min(100, Math.max(0, state.resources[k] + v));
  }

  if (state.isCommander) {
    if (!choice.good) {
      state.commanderMistakes++;
      addLog(`Commander chose a less favorable option (mistakes: ${state.commanderMistakes}/3)`);
    } else {
      addLog(`Commander decided: ${choice.text}`);
    }
    if (lifeCritical && !choice.good) {
      addLog("⚠️ System intervention: To protect human life, the safer option was automatically applied.");
      state.resources.shield = Math.min(100, state.resources.shield + 8);
      state.resources.morale = Math.min(100, state.resources.morale + 3);
    }
  } else {
    addLog(`You voted for: ${choice.text}`);
    if (!choice.good && idx !== majorityIdx) {
      state.dissentCount++;
      addLog(`Your opinion differed from the majority (dissent count: ${state.dissentCount})`);
      if (state.dissentCount >= 3) {
        endMission(false, "You were sent back to Earth for repeatedly disagreeing with the team. Remember: teamwork is critical in space missions.", true);
        return;
      }
    } else {
      state.dissentCount = Math.max(0, state.dissentCount - 0.4);
    }
  }

  document.getElementById('event-area').classList.add('hidden');
  state.paused = false;
  renderBars();
}

// ========== UI helpers ==========
function renderBars() {
  const bars = document.getElementById('bars');
  const labels = { oxygen: 'Life Support O₂', power: 'Power', food: 'Food', shield: 'Radiation Shield', morale: 'Morale', water: 'Water' };
  bars.innerHTML = '';
  for (const [key, val] of Object.entries(state.resources)) {
    const color = val > 60 ? 'var(--ok)' : val > 30 ? 'var(--warn)' : 'var(--danger)';
    bars.innerHTML += `
      <div class="bar-row">
        <div class="bar-label">${labels[key]}</div>
        <div class="bar-bg"><div class="bar-fill" style="width:${Math.max(0,val)}%;background:${color}"></div></div>
        <div style="width:36px;text-align:right;font-size:0.85rem">${Math.round(val)}</div>
      </div>`;
  }
}

function renderFocusButtons() {
  document.getElementById('focus-btns').innerHTML = focuses.map(f => `
    <button onclick="chooseFocus('${f.id}')" ${state.focus === f.id ? 'style="outline:2px solid white"' : ''}>
      ${f.name}<br><small style="opacity:0.8">${f.desc}</small>
    </button>
  `).join('');
}

function addLog(msg) {
  state.log.unshift(`Sol ${state.sol}: ${msg}`);
  if (state.log.length > 25) state.log.pop();
  document.getElementById('log').innerHTML = state.log.map(l => `<div>${l}</div>`).join('');
}

function togglePause() { state.paused = !state.paused; addLog(state.paused ? 'Paused' : 'Resumed'); }
function setSpeed(s) { state.speed = s; }

function endMission(success, reason, isKicked = false) {
  state.running = false;
  clearInterval(state.interval);
  document.getElementById('game-screen').classList.add('hidden');
  document.getElementById('end-screen').classList.remove('hidden');

  document.getElementById('end-title').textContent = isKicked
    ? "🚀 Sent Back to Earth"
    : (success ? "🎉 Mission Success!" : "💥 Mission Failed");

  document.getElementById('end-report').innerHTML = `
    <p><strong>Reason:</strong> ${reason}</p>
    <p>Sols survived: ${state.sol}</p>
    <p>Profession: ${state.profession.name}　Role: ${state.isCommander ? "Commander" : "Crew Member"}</p>
    <p style="margin-top:14px;font-size:0.95rem">
      ${isKicked
        ? "This experience reminds us that in extreme space environments, teamwork and communication are often as important as technical skills."
        : "You experienced role division, voting decisions, and resource trade-offs of a real space mission. Hopefully this simulation helps you better understand the qualities astronauts need."}
    </p>
  `;
}
